# 跟踪数据读取方式

使用 Mybatis Plus 框架的 LambdaWrapper 来读取相关数据库表，Mybatis Plus 功能不能满足的，使用 Mybatis 在 XML 中写 sql 的形式来完成

## Postgres quality 数据库

### qm_mat_cells_blood 机组内物料关系表

该表描述每个加工单元的入口出口物料流转情况，根据 out_mat_no + out_mat_repeat_prod_no + cell_code 可以确定其 in_mat_no + in_mat_repeat_prod_no + cell_code

mat_type、out_mat_sequence_no 用来区分分卷/并卷/正常卷的情况

head_cut_length、tail_cut_length

left_trim_width、right_trim_width

por_type、tr_type

```sql
CREATE TABLE "public"."qm_mat_cells_blood" (
  "id" bigint NOT NULL,
  "deleted" integer,
  "create_time" timestamp(6) without time zone,
  "create_org" bigint,
  "create_user" bigint,
  "update_time" timestamp(6) without time zone,
  "update_user" bigint,
  "backlog_code" character varying,
  "unit_code" character varying,
  "out_mat_no" character varying,
  "out_mat_track_no" character varying,
  "out_mat_prod_no" character varying,
  "in_mat_no" character varying,
  "in_mat_track_no" character varying,
  "in_mat_prod_no" character varying,
  "mat_type" character varying,
  "out_mat_sequence_no" character varying,
  "cell_code" character varying,
  "out_mat_length" numeric,
  "out_mat_thick" numeric,
  "out_mat_width" numeric,
  "start_prod_time" timestamp without time zone,
  "end_prod_time" timestamp without time zone,
  "head_cut_length" numeric,
  "tail_cut_length" numeric,
  "left_trim_width" numeric,
  "right_trim_width" numeric,
  "por_type" character varying,
  "tr_type" character varying,
  "out_mat_repeat_prod_no" character varying,
  "in_mat_repeat_prod_no" character varying,
  PRIMARY KEY ("id")
);

COMMENT ON TABLE "public"."qm_mat_cells_blood" IS '机组内物料关系表';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."id" IS '主键';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."deleted" IS '删除标记';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."create_time" IS '创建时间';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."create_org" IS '创建组织';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."create_user" IS '创建人';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."update_time" IS '更新时间';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."update_user" IS '更新人';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."backlog_code" IS '工序代码';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."unit_code" IS '机组代码';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_no" IS '出口物料号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_track_no" IS '出口物料跟踪号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_prod_no" IS '出口物料跟踪生产次数号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."in_mat_no" IS '入口物料号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."in_mat_track_no" IS '入口物料跟踪号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."in_mat_prod_no" IS '入口物料跟踪生产次数号';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."mat_type" IS '关系类型（0正常，1并卷、2分卷、3混钢坯）';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_sequence_no" IS '出口物料顺序号（分卷情况下子卷的顺序号）';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."cell_code" IS '加工单元代码';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_length" IS '出口钢卷长度m';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_thick" IS '出口钢卷厚度mm';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_width" IS '出口宽度mm';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."start_prod_time" IS '生产开始时间';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."end_prod_time" IS '生产结束时间';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."head_cut_length" IS '切头量m';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."tail_cut_length" IS '切尾量m';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."left_trim_width" IS '切边量-左边mm';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."right_trim_width" IS '切边量-右边mm';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."por_type" IS '开卷方式(0上1下)';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."tr_type" IS '卷取方式(0上1下)';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."out_mat_repeat_prod_no" IS '出口物料重复生产次数号（仅机组内物料关系表使用，与成品卷切片表对接，取数逻辑来自物料档表）';
COMMENT ON COLUMN "public"."qm_mat_cells_blood"."in_mat_repeat_prod_no" IS '入口物料重复生产次数号（仅机组内物料关系表使用，与在线卷跟踪表对接，取数逻辑来自L1入口物料缓存表）';

```
