# 数字钢卷切片表存储细则

使用 Mybatis Plus 框架来写入相关数据库表，Mybatis Plus 功能不能满足的，使用 Mybatis 在 XML 中写 sql 的形式来完成

每个机组有一张对应的数字钢卷切片表，表名为 `qm_dc_<unit_code_lowercase>_pieces`

#### 表结构

该表的工艺参数字段列获取自 cube-api 的/aygg_tracking 跟踪树，每次进行切片计算前，要先调用该接口获取最新的工艺参数字段列，并更新数字钢卷切片表结构（只新增列，不删除列）

```sql
CREATE TABLE "qm_dc_<unit_code_lowercase>_pieces" ("id" bigint NOT NULL,
                                          "unit_code"              CHARACTER VARYING(50),
                                          "in_mat_no"              CHARACTER VARYING(50),
                                          "coiler_method"          CHARACTER VARYING(50),
                                          "coiler_method_name"     CHARACTER VARYING(50),
                                          "coiler_device_code"     CHARACTER VARYING(50),
                                          "coiler_device_name"     CHARACTER VARYING(50),
                                          "coiler_max_length"      CHARACTER VARYING(50),
                                          "create_user"            BIGINT  DEFAULT '1831666618627928065'::bigint,
                                          "update_user"            BIGINT,
                                          "create_time"            TIMESTAMP WITHOUT TIME ZONE,
                                          "update_time"            TIMESTAMP WITHOUT TIME ZONE,
                                          "create_org"             BIGINT,
                                          "deleted"                INTEGER DEFAULT 0,
                                          "uncoiler_method"        CHARACTER VARYING(50),
                                          "uncoiler_method_name"   CHARACTER VARYING(50),
                                          "pass_no"                INTEGER,
                                          "uncoiler_device_code"   CHARACTER VARYING(50),
                                          "uncoiler_device_name"   CHARACTER VARYING(50),
                                          "uncoiler_max_length"    CHARACTER VARYING(50),
                                          "in_mat_repeat_prod_no"  CHARACTER VARYING(50),
                                          "cell_code"              CHARACTER VARYING(50),
                                          PRIMARY KEY ("id"));

COMMENT ON COLUMN "qm_dc_coiler_log"."id" IS '主键';

COMMENT ON COLUMN "qm_dc_coiler_log"."unit_code" IS '机组代码';

COMMENT ON COLUMN "qm_dc_coiler_log"."coiler_method_name" IS '卷取方式名称';

COMMENT ON COLUMN "qm_dc_coiler_log"."coiler_device_code" IS '卷取设备代码';

COMMENT ON COLUMN "qm_dc_coiler_log"."coiler_device_name" IS '卷取设备名称';

COMMENT ON COLUMN "qm_dc_coiler_log"."coiler_max_length" IS '卷取设备最大长度';

COMMENT ON COLUMN "qm_dc_coiler_log"."create_user" IS '创建人';

COMMENT ON COLUMN "qm_dc_coiler_log"."update_user" IS '更新人';

COMMENT ON COLUMN "qm_dc_coiler_log"."create_time" IS '创建时间';

COMMENT ON COLUMN "qm_dc_coiler_log"."update_time" IS '更新时间';

COMMENT ON COLUMN "qm_dc_coiler_log"."create_org" IS '创建组织';

COMMENT ON COLUMN "qm_dc_coiler_log"."deleted" IS '删除标志';

COMMENT ON COLUMN "qm_dc_coiler_log"."uncoiler_method" IS '卷取方式代码';

COMMENT ON COLUMN "qm_dc_coiler_log"."uncoiler_method_name" IS '卷取方式名称';

COMMENT ON COLUMN "qm_dc_coiler_log"."uncoiler_max_length" IS '开卷设备最大长度';

COMMENT ON COLUMN "qm_dc_coiler_log"."in_mat_repeat_prod_no" IS '入口物料重复生产号；历史值从 in_mat_prod_no 复制';

ALTER TABLE "qm_dc_coiler_log" OWNER TO "postgres";
```
